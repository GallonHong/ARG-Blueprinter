# 《受试者脱离守则》ARG 蓝图构建脚本
# 使用方法：全选复制 → 粘贴到 http://localhost:5173 应用的「DSH 联动」面板脚本框点「立即执行」，
# 或粘贴到应用内终端运行；也可以在 DSH 会话里说"把 .cli.txt 的脚本构建到画布"。
touch subject_room -t Desktop -n "受试者房间" --template "调查员工作台" --start
touch rules_v4 -t Browse -n "守则·第4版修订" --template "SCP 绝密卷宗"
touch rules_v3 -t Browse -n "守则·第3版（已废弃）" --template "黑客数据窃密流"
set rules_v3 body="第3版第2条：若在镜中看到自己眨眼，请与它交谈。—— 该条已被第4版删除，删除原因：无记录。"
touch corridor_404 -t Browse -n "无尽晨光快捷酒店·走廊404" --template "极简现代杂志"
touch tiled_basin -t Browse -n "静止游泳馆" --template "遇害者手写日记"
set corridor_404 body="米黄色地毯，消毒水味。所有房门都没有门牌号。走廊尽头有一扇门，门后传来妈妈的声音：'开门，外面冷。'你记得规则9：视线向下。"
set tiled_basin body="淡蓝色瓷砖，无出口。荧光灯发出微弱的蜂鸣。水面下有东西在缓缓上浮——一张巨大的苍白面孔。你没有在溺水，你在被浸泡。"
touch reality_lock -t Login -n "现实检验终端" --template "BIOS 开机验证"
set reality_lock password="cold"
set reality_lock body="系统提示：请输入本周期检验口令。[提示：规则1——醒来后先摸水龙头]"
set reality_lock loginTarget="end_wake"
touch archive_search -t Search -n "梦境档案馆检索" --template "终端搜索"
touch ward_chat -t Chat -n "走廊另一端的信道" --template "暗黑加密通讯"
touch end_wake -t Ending -n "结局·冷水醒来" --template "CRT 黑屏"
touch end_counted -t Ending -n "结局·已被清点" --template "报纸头版通报"
touch end_topo -t Ending -n "结局·拓扑捕获" --template "CRT 黑屏"
touch end_true -t Ending -n "结局·观察者即被困者" --template "案件结案判定书"
ln subject_room rules_v4 -p "守则副本.pdf" --icon "📄"
ln subject_room reality_lock -p "检验终端.exe" --icon "🖥️"
ln subject_room archive_search -p "档案馆.exe" --icon "🔍"
ln subject_room ward_chat -p "床头的电话" --icon "☎️"
ln rules_v4 rules_v3 -p "翻阅修订记录" --icon "📚"
ln rules_v3 rules_v4 -p "返回守则" --icon "📚"
ln corridor_404 subject_room -p "折返" --icon "↩️"
ln tiled_basin subject_room -p "浮上水面" --icon "🌊"
ln archive_search corridor_404 -p "查「走廊404」" --icon "🏨"
ln archive_search tiled_basin -p "查「静止游泳馆」" --icon "🏊"
rule archive_search "第 4 版" rules_v4
rule archive_search "第 3 版" rules_v3
rule archive_search "走廊 404" corridor_404
rule archive_search "游泳馆" tiled_basin
rule archive_search "巡查员" ward_chat
contact ward_chat "家人" --avatar "👨‍👩‍👧" --bio "走廊尽头的声音"
contact ward_chat "巡查员" --avatar "🚶" --bio "无关节，匀速滑行"
msg ward_chat "家人" npc "孩子，开门，外面冷。"
msg ward_chat "巡查员" npc "……正在清点空房间。本层空房间数量：1。"
choice ward_chat "家人" "打开房门" end_counted --reply "门开了。里面只有一台播放雪花屏的电视机，屏幕里映出的是走廊尽头的你。"
choice ward_chat "家人" "检查门牌号" corridor_404 --reply "门牌号：404。没有 403，也没有 405。走廊在变长。"
choice ward_chat "巡查员" "保持视线向下，不出声" subject_room --reply "湿润的拖拽声远去了。你数了数，自己还能睁开眼。"
choice ward_chat "家人" "相信这是现实，跟它走" end_topo --reply "你第 4 次从床上坐起。窗外没有街道，只有不断向下延伸的楼梯。"
choice ward_chat "巡查员" "问它：你们清点的是谁？" end_true --reply "它停住了。缓慢地，一根根关节从它的四肢里长出来。它说：'从你进来那天起，你就在空房间名单上。'"
choice ward_chat "家人" "我摸过水龙头了，是冷的——我不会开门" end_wake --reply "妈妈的声音消失了。你醒在真实的床上，水龙头正滴着冷水。可你突然想起：守则里没有写'水龙头在哪里'。"
